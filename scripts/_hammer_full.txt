_ds = cffi.Session(impersonate='chrome110')

def fetch_proxy(region):
    for _ in range(3):
        try:
            r = _ds.get(KOOKEEY_API.format(region=region), timeout=15)
            parts = r.text.strip().split(':')
            if len(parts) == 4:
                return f'http://{parts[2]}:{parts[3]}@{parts[0]}:{parts[1]}'
        except Exception:
            pass
        time.sleep(1)
    raise ValueError('proxy fail: ' + region)

def make_session(proxy_url):
    return cffi.Session(impersonate='chrome110', proxy=proxy_url, curl_options={CurlOpt.PRE_PROXY: PRE_PROXY})

def find_redirect(d):
    if not isinstance(d, dict):
        return ''
    def _s(v):
        if isinstance(v, str):
            m = PM_REDIRECT_RE.search(v)
            return m.group(0) if m else ''
        if isinstance(v, dict):
            na = v.get('next_action')
            if isinstance(na, dict) and na.get('type') == 'redirect_to_url':
                rtu = na.get('redirect_to_url')
                if isinstance(rtu, dict):
                    url = str(rtu.get('url') or '').strip()
                    if url:
                        return url
            for x in v.values():
                r = _s(x)
                if r:
                    return r
        if isinstance(v, list):
            for x in v:
                r = _s(x)
                if r:
                    return r
        return ''
    for k in ('setup_intent', 'payment_intent'):
        r = _s(d.get(k) or {})
        if r:
            return r
    return _s(d)

            if any(x in low for x in ('tls', 'timeout', 'proxy', 'connect', 'ssl')):
                res['approve_result'] = 'network_error'
    return res

def main():
    print(f'=== multi-IP hammer: proxy={PROXY_REGION} checkout={CHECKOUT_COUNTRY}/{CHECKOUT_CURRENCY} max={MAX_ATTEMPTS} stop_on_first={STOP_ON_FIRST} ===', flush=True)
    results = []
    for i in range(1, MAX_ATTEMPTS + 1):
        r = try_once(i, CHECKOUT_COUNTRY, CHECKOUT_CURRENCY, PROXY_REGION)
        results.append(r)
        print(
            f'  [{i:02d}] approve={str(r.get("approve_result","")):16s} ba_ok={str(r.get("ba_ok")):5s} amt={str(r.get("amt","")):6s} ba={r.get("ba","")} err={str(r.get("err",""))[:50]}',
            flush=True,
        )
        if r.get('ba_ok') and STOP_ON_FIRST:
            print(f'  BA_URL={r.get("ba_url","")}', flush=True)
            break
        time.sleep(0.5)

    print('\n=== SUMMARY ===', flush=True)
    cnt = Counter(r.get('approve_result') or 'empty' for r in results)
    for k, v in cnt.most_common():
        ok = sum(1 for r in results if (r.get('approve_result') or 'empty') == k and r.get('ba_ok'))
        print(f'  {k}: {v} total, {ok} ba_ok', flush=True)
    ba_ok = sum(1 for r in results if r.get('ba_ok'))
    print(f'  overall_ba_ok={ba_ok}/{len(results)}', flush=True)

    out = 'scripts/ba_multi_ip_hammer_results.json'
    with open(out, 'w', encoding='utf-8') as f:
        json.dump(results, f, ensure_ascii=False, indent=2)
    print(f'  saved={out}', flush=True)
    print('=== DONE ===', flush=True)

if __name__ == '__main__':
    main()
